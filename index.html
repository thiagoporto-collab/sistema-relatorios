// ========================================== // BACKEND: Google Apps Script (API JSONP) // ========================================== function setupSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const abas = [ 'nomes', 'Periodos', 'Gestores', 'Datas', 'StatusDiario', 'Registros', 'Relatorios', 'Contratos_ListaGestores', 'Contratos_Datas', 'Contratos_Gestores', 'Contratos_Colaboradores', 'Contratos_Registros' ]; abas.forEach(aba => { let sheet = getSheetSafe(aba); if (sheet.getLastRow() === 0) { if(aba === 'nomes') sheet.appendRow(['MATRÍCULA', 'NOME', 'Código de Barras']); if(aba === 'Periodos') sheet.appendRow(['ID', 'DataInicial', 'DataFinal']); if(aba === 'Gestores') sheet.appendRow(['ID', 'Nome', 'Turno']); if(aba === 'Datas') sheet.appendRow(['ID', 'ID_Periodo', 'DataString', 'DataISO', 'Turno']); if(aba === 'StatusDiario') sheet.appendRow(['ID_Data', 'ID_Gestor', 'TeveRelatorio']); if(aba === 'Registros') sheet.appendRow(['ID_Data', 'ID_Gestor', 'Acao', 'CodigoLido', 'AdminLogado', 'DataHora']); if(aba === 'Relatorios') sheet.appendRow([ 'PeriodoID', 'Data', 'Turno', 'Gestor', 'Status', 'AdminEntregou', 'ParaQuemEntregou', 'DataEntrega', 'QuemDevolveu', 'AdminRecebeuDevolucao', 'DataDevolucao' ]); if(aba === 'Contratos_ListaGestores') sheet.appendRow(['ID', 'Nome']); if(aba === 'Contratos_Datas') sheet.appendRow(['ID', 'DataString', 'DataISO']); if(aba === 'Contratos_Gestores') sheet.appendRow(['ID_Data', 'ID_Gestor']); if(aba === 'Contratos_Colaboradores') sheet.appendRow(['ID_Data', 'ID_Gestor', 'Matricula']); if(aba === 'Contratos_Registros') sheet.appendRow(['ID_Data', 'ID_Gestor', 'Matricula', 'Acao', 'CodigoLido', 'AdminLogado', 'DataHora']); } }); } function doGet(e) { try { var callback = e.parameter.callback || 'callback'; var action = e.parameter.action; var payload = e.parameter.payload ? JSON.parse(e.parameter.payload) : {}; var responseData = null; if (action === 'sincronizar') { responseData = syncData(); } else if (action === 'registrarAcao') { responseData = registrarAcao(payload); } else if (action === 'salvarPeriodo') { responseData = salvarPeriodoLogica(payload); } else if (action === 'excluirPeriodo') { responseData = deletarLinha('Periodos', payload.id); } else if (action === 'salvarGestor') { responseData = salvarGestorLogica(payload); } else if (action === 'excluirGestor') { responseData = deletarLinha('Gestores', payload.id); } else if (action === 'salvarStatusDiario') { responseData = salvarStatusDiarioLogica(payload); } else if (action === 'salvarGestorContrato') { responseData = salvarGestorContratoLogica(payload); } else if (action === 'excluirGestorContrato') { responseData = deletarLinha('Contratos_ListaGestores', payload.id); } else if (action === 'salvarDataContrato') { responseData = salvarDataContratoLog(payload); } else if (action === 'salvarGestoresContrato') { responseData = salvarGestoresContratoLog(payload); } else if (action === 'addColabContrato') { responseData = salvarData('Contratos_Colaboradores', [payload.idData, payload.idGestor, payload.matricula]); } else if (action === 'removerColabContrato') { responseData = deletarColabContrato(payload); } else if (action === 'registrarAcaoContrato') { let admin = buscarPessoa(payload.responsavel); let nomeAdmin = admin.matricula !== 'N/A' ? admin.nome : payload.responsavel; responseData = salvarData('Contratos_Registros', [payload.idData, payload.idGestor, payload.matricula, payload.acao, payload.codigoLido, nomeAdmin, payload.dataHora]); } else if (action === 'registrarAcaoContratoColetivo') { let sheet = getSheetSafe('Contratos_Registros'); payload.registros.forEach(reg => { let admin = buscarPessoa(reg.responsavel); let nomeAdmin = admin.matricula !== 'N/A' ? admin.nome : reg.responsavel; sheet.appendRow([reg.ID_Data, reg.ID_Gestor, reg.Matricula, reg.Acao, reg.CodigoLido, nomeAdmin, reg.DataHora]); }); responseData = { success: true }; } var output = callback + "(" + JSON.stringify({ status: 'success', data: responseData }) + ");"; return ContentService.createTextOutput(output).setMimeType(ContentService.MimeType.JAVASCRIPT); } catch (error) { var callback = e.parameter ? e.parameter.callback : 'callback'; var output = callback + "(" + JSON.stringify({ status: 'error', message: error.toString() }) + ");"; return ContentService.createTextOutput(output).setMimeType(ContentService.MimeType.JAVASCRIPT); } } function getSheetSafe(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(name); if (!sheet) sheet = ss.insertSheet(name); return sheet; } function syncData() { const getRows = (sheetName) => { let sheet = getSheetSafe(sheetName); let data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; let headers = data.shift(); return data.map(row => { let obj = {}; headers.forEach((h, i) => { let val = row[i]; if (val instanceof Date) { if (h === 'DataISO' || h === 'DataInicial' || h === 'DataFinal') { val = Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd"); } else { val = Utilities.formatDate(val, Session.getScriptTimeZone(), "dd/MM/yyyy"); } } obj[h] = val; }); return obj; }); }; return { periodos: getRows('Periodos'), gestores: getRows('Gestores').sort((a,b) => (a.Nome || '').localeCompare(b.Nome || '')), datas: getRows('Datas'), statusDiario: getRows('StatusDiario'), registros: getRows('Registros'), nomes: getRows('nomes'), contratosListaGestores: getRows('Contratos_ListaGestores').sort((a,b) => (a.Nome || '').localeCompare(b.Nome || '')), contratosDatas: getRows('Contratos_Datas'), contratosGestores: getRows('Contratos_Gestores'), contratosColab: getRows('Contratos_Colaboradores'), contratosRegistros: getRows('Contratos_Registros') }; } function salvarPeriodoLogica(payload) { let sheet = getSheetSafe('Periodos'); if(payload.isEdit) { let data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === payload.id) { sheet.getRange(i + 1, 2).setValue(payload.inicio); sheet.getRange(i + 1, 3).setValue(payload.fim); return { success: true }; } } } sheet.appendRow([payload.id, payload.inicio, payload.fim]); return { success: true }; } function salvarGestorLogica(payload) { let sheet = getSheetSafe('Gestores'); let data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === payload.id) { sheet.getRange(i + 1, 2).setValue(payload.nome); sheet.getRange(i + 1, 3).setValue(payload.turno); return { success: true }; } } sheet.appendRow([payload.id, payload.nome, payload.turno]); return { success: true }; } function salvarGestorContratoLogica(payload) { let sheet = getSheetSafe('Contratos_ListaGestores'); let data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === payload.id) { sheet.getRange(i + 1, 2).setValue(payload.nome); return { success: true }; } } sheet.appendRow([payload.id, payload.nome]); return { success: true }; } function salvarData(sheetName, rowData) { getSheetSafe(sheetName).appendRow(rowData); return { success: true }; } function deletarLinha(sheetName, id) { let sheet = getSheetSafe(sheetName); let data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] == id) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false }; } function salvarStatusDiarioLogica(payload) { let sheetDatas = getSheetSafe('Datas'); let sheetStatus = getSheetSafe('StatusDiario'); if(payload.isEdit) { let datas = sheetDatas.getDataRange().getValues(); for (let i = 1; i < datas.length; i++) { if (datas[i][0] === payload.idData) { sheetDatas.getRange(i + 1, 3).setValue(payload.dataString); sheetDatas.getRange(i + 1, 4).setValue(payload.dataISO); sheetDatas.getRange(i + 1, 5).setValue(payload.turno); break; } } let status = sheetStatus.getDataRange().getValues(); for (let i = status.length - 1; i >= 1; i--) { if (status[i][0] === payload.idData) { sheetStatus.deleteRow(i + 1); } } } else { sheetDatas.appendRow([payload.idData, payload.idPeriodo, payload.dataString, payload.dataISO, payload.turno]); } payload.gestoresStatus.forEach(gs => { sheetStatus.appendRow([payload.idData, gs.idGestor, gs.teve]); }); return { success: true }; } function buscarPessoa(codigo) { if (!codigo) return { matricula: 'N/A', nome: 'Não Identificado' }; let sheet = getSheetSafe('nomes'); let data = sheet.getDataRange().getValues(); let target = String(codigo).trim().toLowerCase(); for(let i = 1; i < data.length; i++) { let cellCode = String(data[i][2]).trim(); let cellMat = String(data[i][0]).trim(); if(cellCode.toLowerCase() === target || cellCode.replace(/\D/g, '') === target.replace(/\D/g, '') || cellMat.replace(/\D/g, '') === target.replace(/\D/g, '')) { return { matricula: cellMat, nome: String(data[i][1]).trim() }; } } return { matricula: 'N/A', nome: String(codigo).trim() }; } function registrarAcao(payload) { let sheetReg = getSheetSafe('Registros'); let dataReg = sheetReg.getDataRange().getValues(); for(let i = 1; i < dataReg.length; i++){ if(dataReg[i][0] === payload.idData && dataReg[i][1] === payload.idGestor && dataReg[i][2] === payload.acao) { return { success: true, message: "Ação já existia no banco." }; } } let adminObj = buscarPessoa(payload.responsavel); let nomeAdminParaSalvar = adminObj.matricula !== 'N/A' ? adminObj.nome : payload.responsavel; sheetReg.appendRow([payload.idData, payload.idGestor, payload.acao, payload.codigoLido, nomeAdminParaSalvar, payload.dataHora]); atualizarAbaRelatorios(payload, nomeAdminParaSalvar); return { success: true }; } function atualizarAbaRelatorios(payload, nomeAdminSalvo) { let sheetRel = getSheetSafe('Relatorios'); let sheetDatas = getSheetSafe('Datas'); let datas = sheetDatas.getDataRange().getValues(); let periodoId = ''; let dataStr = ''; for (let i = 1; i < datas.length; i++) { if (String(datas[i][0]) === String(payload.idData)) { periodoId = datas[i][1]; dataStr = datas[i][2]; break; } } let gestorNome = payload.gestorNome || 'Desconhecido'; let pessoaLida = buscarPessoa(payload.codigoLido); let strPessoaLida = pessoaLida.matricula !== 'N/A' ? `${pessoaLida.nome}` : `${payload.codigoLido}`; let turno = payload.turno || 'A'; let rows = sheetRel.getDataRange().getValues(); let foundIndex = -1; for (let k = 1; k < rows.length; k++) { if (String(rows[k][0]) === String(periodoId) && String(rows[k][1]) === String(dataStr) && String(rows[k][3]).trim() === String(gestorNome).trim()) { foundIndex = k + 1; break; } } if (foundIndex > -1) { if (payload.acao === 'Entregar') { sheetRel.getRange(foundIndex, 5).setValue('Entregue'); sheetRel.getRange(foundIndex, 6).setValue(nomeAdminSalvo); sheetRel.getRange(foundIndex, 7).setValue(strPessoaLida); sheetRel.getRange(foundIndex, 8).setValue(payload.dataHora); } else if (payload.acao === 'Devolver') { sheetRel.getRange(foundIndex, 5).setValue('Devolvido'); sheetRel.getRange(foundIndex, 9).setValue(strPessoaLida); sheetRel.getRange(foundIndex, 10).setValue(nomeAdminSalvo); sheetRel.getRange(foundIndex, 11).setValue(payload.dataHora); } } else { let status = payload.acao === 'Entregar' ? 'Entregue' : 'Devolvido'; let admEntregou = payload.acao === 'Entregar' ? nomeAdminSalvo : ''; let paraQuemEntregou = payload.acao === 'Entregar' ? strPessoaLida : ''; let dtEntrega = payload.acao === 'Entregar' ? payload.dataHora : ''; let qmDevolveu = payload.acao === 'Devolver' ? strPessoaLida : ''; let admRecebeu = payload.acao === 'Devolver' ? nomeAdminSalvo : ''; let dtDevolucao = payload.acao === 'Devolver' ? payload.dataHora : ''; sheetRel.appendRow([periodoId, dataStr, turno, gestorNome, status, admEntregou, paraQuemEntregou, dtEntrega, qmDevolveu, admRecebeu, dtDevolucao]); } } function salvarDataContratoLog(payload) { let sheet = getSheetSafe('Contratos_Datas'); if(payload.isEdit) { let data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === payload.id) { sheet.getRange(i + 1, 2).setValue(payload.dataStr); sheet.getRange(i + 1, 3).setValue(payload.dataISO); return { success: true }; } } } sheet.appendRow([payload.id, payload.dataStr, payload.dataISO]); return { success: true }; } function salvarGestoresContratoLog(payload) { let sheet = getSheetSafe('Contratos_Gestores'); let data = sheet.getDataRange().getValues(); for (let i = data.length - 1; i >= 1; i--) { if (data[i][0] === payload.idData) sheet.deleteRow(i + 1); } payload.gestores.forEach(gId => sheet.appendRow([payload.idData, gId])); return { success: true }; } function deletarColabContrato(payload) { let sheet = getSheetSafe('Contratos_Colaboradores'); let data = sheet.getDataRange().getValues(); for (let i = data.length - 1; i >= 1; i--) { if (data[i][0] === payload.idData && data[i][1] === payload.idGestor && String(data[i][2]) === String(payload.matricula)) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false }; }
